        // 1. Tactile Ripple and Click Effect
        document.addEventListener('mousedown', function (e) {
            const el = e.target.closest('a, button, .translate-box, .info-item');
            if (!el) return;
            el.style.transform = "scale(0.95)";
            setTimeout(() => { el.style.transform = ""; }, 150);
            const rect = el.getBoundingClientRect();
            const ripple = document.createElement('span');
            ripple.className = 'ripple';
            const size = Math.max(rect.width, rect.height);
            ripple.style.width = ripple.style.height = size + 'px';
            ripple.style.left = (e.clientX - rect.left - size/2) + 'px';
            ripple.style.top = (e.clientY - rect.top - size/2) + 'px';
            el.appendChild(ripple);
            setTimeout(() => ripple.remove(), 600);
        });

        // 2. Language Logic
        function toggleLang() { const l = document.getElementById('langList'); l.style.display = (l.style.display === 'grid') ? 'none' : 'grid'; }

            const combo = document.querySelector(".goog-te-combo");
            if (combo) { combo.value = code; combo.dispatchEvent(new Event('change')); toggleLang(); }
            else { alert("Engine loading..."); }
        }
